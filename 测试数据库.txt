-- 创建业务数据库
DROP DATABASE IF EXISTS business_db;
CREATE DATABASE IF NOT EXISTS business_db;
USE business_db;


-- 2.1 用户表(users) - 对应PDF表2.1
CREATE TABLE IF NOT EXISTS users (
    user_id VARCHAR(20) NOT NULL PRIMARY KEY COMMENT '用户ID',
    his_num VARCHAR(20) NOT NULL COMMENT 'HIS代码',
    username VARCHAR(20) NOT NULL COMMENT '医生姓名',
    password VARCHAR(10) COMMENT '密码',
    dept_name VARCHAR(20) COMMENT '所属科室',
    title VARCHAR(20) COMMENT '职称',
    speciality VARCHAR(255) COMMENT '擅长',
    introduce VARCHAR(255) COMMENT '简介',
    online_time DATETIME COMMENT '上线时间',
    collection INT DEFAULT 0 COMMENT '收藏数',
    consultation INT DEFAULT 0 COMMENT '咨询数',
    drug_flag CHAR(2) COMMENT '是否线上处方',
    lab_flag CHAR(2) COMMENT '是否线上检验',
    exam_flag CHAR(2) COMMENT '是否线上检查',
    CA_flag CHAR(2) COMMENT '是否CA认证',
    message_way CHAR(2) COMMENT '消息推送方式',
    examine_flag CHAR(2) DEFAULT '1' COMMENT '审核标识：1通过，0不通过',
    examine_oper VARCHAR(10) COMMENT '审核人员',
    -- 为username创建索引，用于外键引用
    INDEX idx_username (username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='用户表';

-- 2.3 患者表(patient) - 对应PDF表2.3
CREATE TABLE IF NOT EXISTS patient (
    patient_id VARCHAR(20) NOT NULL PRIMARY KEY COMMENT '患者ID',
    patient_card VARCHAR(20) COMMENT '就诊卡号',
    patient_name VARCHAR(20) NOT NULL COMMENT '患者姓名',
    patient_age INT COMMENT '患者年龄',
    patient_sex CHAR(2) COMMENT '性别：男1，女2',
    patient_mobile VARCHAR(11) COMMENT '手机号',
    patient_address VARCHAR(255) COMMENT '地址',
    enable_flag CHAR(1) NOT NULL DEFAULT '1' COMMENT '启用状态：0禁用，1启用',
    create_time DATETIME NOT NULL COMMENT '创建时间',
    -- 为patient_name创建索引
    INDEX idx_patient_name (patient_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='患者表';


-- 2.6 问诊记录主表(clinic_master) - 对应PDF表2.6
CREATE TABLE IF NOT EXISTS clinic_master (
    clinic_id VARCHAR(20) NOT NULL PRIMARY KEY COMMENT '问诊序号',
    clinic_date DATETIME NOT NULL COMMENT '问诊日期',
    patient_id VARCHAR(20) NOT NULL COMMENT '患者ID',
    patient_name VARCHAR(20) NOT NULL COMMENT '患者姓名',
    patient_age INT COMMENT '患者年龄',
    patient_sex CHAR(8) COMMENT '患者性别',
    username VARCHAR(20) NOT NULL COMMENT '问诊医生',
    type_code CHAR(2) COMMENT '问诊方式：1图文，2电话，3视频',
    clinic_type VARCHAR(20) COMMENT '问诊方式表号类',
    -- 创建索引
    INDEX idx_patient_id (patient_id),
    INDEX idx_username (username),
    INDEX idx_clinic_date (clinic_date),
    -- 外键约束
    CONSTRAINT fk_master_patient FOREIGN KEY (patient_id) REFERENCES patient(patient_id),
    CONSTRAINT fk_master_user FOREIGN KEY (username) REFERENCES users(username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='问诊记录主表';

-- 2.7 门诊医嘱主记录表(clinic_orders) - 对应PDF表2.7
CREATE TABLE IF NOT EXISTS clinic_orders (
    order_id VARCHAR(20) NOT NULL PRIMARY KEY COMMENT '门诊序号',
    clinic_id VARCHAR(20) NOT NULL COMMENT '问诊序号',
    serial_no VARCHAR(10) NOT NULL UNIQUE COMMENT '流水号',
    patient_name VARCHAR(20) NOT NULL COMMENT '患者姓名',
    patient_card VARCHAR(20) COMMENT '就诊卡号',
    visit_date DATETIME COMMENT '就诊日期',
    order_info VARCHAR(255) COMMENT '医嘱建议',
    dept_name VARCHAR(20) COMMENT '开单科室',
    username VARCHAR(20) NOT NULL COMMENT '开单医生',
    order_date DATETIME COMMENT '开单日期',
    charge DECIMAL(10,4) COMMENT '支付金额',
    pass_flag CHAR(2) DEFAULT '1' COMMENT '咨询是否有效：1有效，0无效',
    invalid_reason VARCHAR(100) COMMENT '无效原因',
    reply_date INT COMMENT '医生回复时长(分钟)',
    clinic_state CHAR(8) DEFAULT '1' COMMENT '状态：1新申请，2接收，3结束',
    cancel_flag CHAR(1) DEFAULT '0' COMMENT '取消标记：0未取消，1已取消',
    cancel_reason VARCHAR(255) COMMENT '取消原因',
    cancel_time DATETIME COMMENT '取消时间',
    -- 创建索引
    INDEX idx_clinic_id (clinic_id),
    INDEX idx_username (username),
    INDEX idx_serial_no (serial_no),
    INDEX idx_clinic_state (clinic_state),
    INDEX idx_pass_flag (pass_flag),
    -- 外键约束
    CONSTRAINT fk_orders_master FOREIGN KEY (clinic_id) REFERENCES clinic_master(clinic_id),
    CONSTRAINT fk_orders_user FOREIGN KEY (username) REFERENCES users(username)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='门诊医嘱主记录表';

-- 2.14 黑名单表(blacklist) - 对应PDF表2.14
CREATE TABLE IF NOT EXISTS blacklist (
    black_id VARCHAR(20) NOT NULL PRIMARY KEY COMMENT '序号',
    user_id VARCHAR(20) NOT NULL COMMENT '用户ID(医生)',
    patient_id VARCHAR(20) NOT NULL COMMENT '患者ID',
    reason VARCHAR(255) COMMENT '加入原因',
    create_time DATETIME NOT NULL COMMENT '创建时间',
    enable_flag CHAR(1) NOT NULL DEFAULT '0' COMMENT '解除标记：0未解除，1已解除',
    -- 创建索引
    INDEX idx_user_id (user_id),
    INDEX idx_patient_id (patient_id),
    INDEX idx_enable_flag (enable_flag),
    -- 外键约束
    CONSTRAINT fk_blacklist_user FOREIGN KEY (user_id) REFERENCES users(user_id),
    CONSTRAINT fk_blacklist_patient FOREIGN KEY (patient_id) REFERENCES patient(patient_id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='黑名单表';
