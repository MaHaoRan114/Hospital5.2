-- ==================== 使用 JKDB 数据库 ====================
USE JKDB;
GO

-- ==================== 删除已存在的表（按依赖顺序） ====================
IF EXISTS (SELECT * FROM sys.objects WHERE name = 'blacklist' AND type = 'U')
    DROP TABLE blacklist;
IF EXISTS (SELECT * FROM sys.objects WHERE name = 'clinic_orders' AND type = 'U')
    DROP TABLE clinic_orders;
IF EXISTS (SELECT * FROM sys.objects WHERE name = 'clinic_master' AND type = 'U')
    DROP TABLE clinic_master;
IF EXISTS (SELECT * FROM sys.objects WHERE name = 'patient' AND type = 'U')
    DROP TABLE patient;
IF EXISTS (SELECT * FROM sys.objects WHERE name = 'users' AND type = 'U')
    DROP TABLE users;
GO

-- ==================== 1. 用户表(users) ====================
CREATE TABLE users (
    user_id VARCHAR(20) NOT NULL PRIMARY KEY,
    his_num VARCHAR(20) NOT NULL,
    username VARCHAR(20) NOT NULL UNIQUE,  -- UNIQUE 约束，供外键引用
    password VARCHAR(10) NULL,
    dept_name VARCHAR(20) NULL,
    title VARCHAR(20) NULL,
    speciality VARCHAR(255) NULL,
    introduce VARCHAR(255) NULL,
    online_time DATETIME NULL,
    collection INT DEFAULT 0,
    consultation INT DEFAULT 0,
    drug_flag CHAR(2) NULL,
    lab_flag CHAR(2) NULL,
    exam_flag CHAR(2) NULL,
    CA_flag CHAR(2) NULL,
    message_way CHAR(2) NULL,
    examine_flag CHAR(2) DEFAULT '1',
    examine_oper VARCHAR(10) NULL
);
GO

CREATE INDEX idx_users_username ON users(username);
GO

-- ==================== 2. 患者表(patient) ====================
CREATE TABLE patient (
    patient_id VARCHAR(20) NOT NULL PRIMARY KEY,
    patient_card VARCHAR(20) NULL,
    patient_name VARCHAR(20) NOT NULL,
    patient_age INT NULL,
    patient_sex CHAR(2) NULL,
    patient_mobile VARCHAR(11) NULL,
    patient_address VARCHAR(255) NULL,
    enable_flag CHAR(1) NOT NULL DEFAULT '1',
    create_time DATETIME NOT NULL
);
GO

CREATE INDEX idx_patient_name ON patient(patient_name);
GO

-- ==================== 3. 问诊记录主表(clinic_master) ====================
CREATE TABLE clinic_master (
    clinic_id VARCHAR(20) NOT NULL PRIMARY KEY,
    clinic_date DATETIME NOT NULL,
    patient_id VARCHAR(20) NOT NULL,
    patient_name VARCHAR(20) NOT NULL,
    patient_age INT NULL,
    patient_sex CHAR(8) NULL,
    username VARCHAR(20) NOT NULL,
    type_code CHAR(2) NULL,
    clinic_type VARCHAR(20) NULL
);
GO

CREATE INDEX idx_master_patient_id ON clinic_master(patient_id);
CREATE INDEX idx_master_username ON clinic_master(username);
GO

-- 外键约束
ALTER TABLE clinic_master ADD CONSTRAINT fk_master_patient 
    FOREIGN KEY (patient_id) REFERENCES patient(patient_id);
ALTER TABLE clinic_master ADD CONSTRAINT fk_master_user 
    FOREIGN KEY (username) REFERENCES users(username);
GO

-- ==================== 4. 门诊医嘱主记录表(clinic_orders) ====================
CREATE TABLE clinic_orders (
    order_id VARCHAR(20) NOT NULL PRIMARY KEY,
    clinic_id VARCHAR(20) NOT NULL,
    serial_no VARCHAR(10) NOT NULL UNIQUE,
    patient_name VARCHAR(20) NOT NULL,
    patient_card VARCHAR(20) NULL,
    visit_date DATETIME NULL,
    order_info VARCHAR(255) NULL,
    dept_name VARCHAR(20) NULL,
    username VARCHAR(20) NOT NULL,
    order_date DATETIME NULL,
    charge DECIMAL(10,4) NULL,
    pass_flag CHAR(2) DEFAULT '1',
    invalid_reason VARCHAR(100) NULL,
    reply_date INT NULL,
    clinic_state CHAR(8) DEFAULT '1',
    cancel_flag CHAR(1) DEFAULT '0',
    cancel_reason VARCHAR(255) NULL,
    cancel_time DATETIME NULL
);
GO

CREATE INDEX idx_orders_clinic_id ON clinic_orders(clinic_id);
CREATE INDEX idx_orders_username ON clinic_orders(username);
CREATE INDEX idx_orders_serial_no ON clinic_orders(serial_no);
GO

ALTER TABLE clinic_orders ADD CONSTRAINT fk_orders_master 
    FOREIGN KEY (clinic_id) REFERENCES clinic_master(clinic_id);
ALTER TABLE clinic_orders ADD CONSTRAINT fk_orders_user 
    FOREIGN KEY (username) REFERENCES users(username);
GO

-- ==================== 5. 黑名单表(blacklist) ====================
CREATE TABLE blacklist (
    black_id VARCHAR(20) NOT NULL PRIMARY KEY,
    user_id VARCHAR(20) NOT NULL,
    patient_id VARCHAR(20) NOT NULL,
    reason VARCHAR(255) NULL,
    create_time DATETIME NOT NULL,
    enable_flag CHAR(1) NOT NULL DEFAULT '0'
);
GO

CREATE INDEX idx_blacklist_user_id ON blacklist(user_id);
CREATE INDEX idx_blacklist_patient_id ON blacklist(patient_id);
GO

ALTER TABLE blacklist ADD CONSTRAINT fk_blacklist_user 
    FOREIGN KEY (user_id) REFERENCES users(user_id);
ALTER TABLE blacklist ADD CONSTRAINT fk_blacklist_patient 
    FOREIGN KEY (patient_id) REFERENCES patient(patient_id);
GO
